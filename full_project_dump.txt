import React, { useState, useEffect } from 'react';
import { Routes, Route, Navigate, Outlet } from 'react-router-dom';
import './App.css';
import { userApi, authApi } from './utils/api';
import { User } from './utils/types';
import { Navbar } from './components/Navbar';
import { LoginView } from './components/LoginView';
import { RegisterView } from './components/RegisterView';
import { EquipmentTab } from './components/EquipmentTab';
import { RequestsTab } from './components/RequestsTab';
import { OperationsTab } from './components/OperationsTab';
import { DirectoryTab } from './components/DirectoryTab';
import { UsersTab } from './components/UsersTab';
import { NotFoundView } from './components/NotFoundView';

function ProtectedLayout({ user, onLogout }: { user: User | null; onLogout: () => void }) {
  if (!user) {
    return <Navigate to="/login" replace />;
  }

  return (
    <>
      <Navbar user={user} onLogout={onLogout} />
      <main>
        <Outlet />
      </main>
    </>
  );
}

function App() {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState<boolean>(true);

  useEffect(() => {
    userApi.getMe()
      .then((data) => setUser(data))
      .catch(() => setUser(null))
      .finally(() => setLoading(false));
  }, []);

  const handleLogout = async () => {
    try {
      await authApi.logout();
    } catch (e) {
      // Игнорируем ошибки сети при выходе
    } finally {
      setUser(null);
    }
  };

  if (loading) {
    return <div style={{ padding: 32, textAlign: 'center' }}>Загрузка сессии...</div>;
  }

  return (
    <Routes>
      {/* Маршруты авторизации: если уже вошел — отправляем к оборудованию */}
      <Route
        path="/login"
        element={user ? <Navigate to="/equipment" replace /> : <LoginView onSuccess={setUser} />}
      />
      <Route
        path="/register"
        element={user ? <Navigate to="/equipment" replace /> : <RegisterView onSuccess={setUser} />}
      />

      {/* Защищенная зона приложения */}
      {user ? (
        <Route element={<ProtectedLayout user={user} onLogout={handleLogout} />}>
          <Route index element={<Navigate to="/equipment" replace />} />
          <Route path="equipment" element={<EquipmentTab user={user} />} />
          <Route path="requests" element={<RequestsTab user={user} />} />
          <Route path="operations" element={<OperationsTab user={user} />} />
          <Route path="directory" element={<DirectoryTab user={user} />} />
          {user.role === 'admin' && (
            <Route path="users" element={<UsersTab currentUser={user} />} />
          )}
          {/* Кастомная страница 404 для авторизованного пользователя с учетом его роли */}
          <Route path="*" element={<NotFoundView user={user} />} />
        </Route>
      ) : (
        /* Если не авторизован — любой неизвестный URL отправляет на /login */
        <Route path="*" element={<Navigate to="/login" replace />} />
      )}
    </Routes>
  );
}

export default App;

export const EquipmentTab: React.FC<{ user: User }> = ({ user }) => {

  const [locations, setLocations] = useState<LocationItem[]>([]);
  const [search, setSearch] = useState('');
  const [editingId, setEditingId] = useState<number | null>(null);

  const [form, setForm] = useState({
    invNumber: '',
    serialNumber: '',
    name: '',
    categoryId: 0,
    locationId: 0,
    purchaseCost: 0,
    status: 'Работает' as EquipmentStatus,
  });

  const canManage = user.role === 'manager' || user.role === 'admin';

  const loadData = async () => {
    try {
      const [eq, cat, loc] = await Promise.all([
        equipmentApi.getAll(),
        directoryApi.getCategories(),
        directoryApi.getLocations()
      ]);
      setList(eq);
      setCategories(cat);
      setLocations(loc);
    } catch (e: any) {
      alert(e.message);
    }
  };

  useEffect(() => { loadData(); }, []);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      const payload = {
        ...form,
        categoryId: form.categoryId ? Number(form.categoryId) : null,
        locationId: form.locationId ? Number(form.locationId) : null,
        purchaseCost: Number(form.purchaseCost),
      };
      if (editingId) {
        await equipmentApi.update(editingId, payload);
      } else {
        await equipmentApi.create(payload);
      }
      resetForm();
      loadData();
    } catch (err: any) {
      alert(err.message);
    }
  };

  const handleEdit = (item: Equipment) => {
    setEditingId(item.id);
    setForm({
      invNumber: item.invNumber,
      serialNumber: item.serialNumber || '',
      name: item.name,
      categoryId: item.categoryId || 0,
      locationId: item.locationId || 0,
      purchaseCost: item.purchaseCost || 0,
      status: item.status,
    });
  };

  const handleDelete = async (id: number) => {
    if (!window.confirm('Удалить оборудование?')) return;
    try {
      await equipmentApi.delete(id);
      loadData();
    } catch (err: any) {
      alert(err.message);
    }
  };

  const resetForm = () => {
    setEditingId(null);
    setForm({
      invNumber: '', serialNumber: '', name: '',
      categoryId: 0, locationId: 0, purchaseCost: 0, status: 'Работает'
    });
  };

  const filtered = list.filter((e) =>
    `${e.invNumber} ${e.name} ${e.categoryName || ''} ${e.locationName || ''}`
      .toLowerCase()
      .includes(search.toLowerCase())
  );

// Общий метод запросов с прозрачным повтором при истечении токена
async function request<T>(url: string, options: RequestInit = {}, isRetry = false): Promise<T> {
  const config: RequestInit = {
    ...options,
    credentials: 'include',
    headers: {
      'Content-Type': 'application/json',
      ...options.headers,
    },
  };

  const response = await fetch(url, config);

  // Если сервер вернул 401 и это не попытка входа/логаута — пробуем обновить токен и повторить запрос
  if (response.status === 401 && !isRetry && !url.startsWith('/auth/')) {
    try {
      const refreshRes = await fetch('/auth/refresh', { method: 'POST', credentials: 'include' });
      if (refreshRes.ok) {
        return await request<T>(url, options, true);
      }
    } catch (e) {
      // Ошибка обновления токена
    }
  }

  const data = await response.json().catch(() => null);

  if (!response.ok) {
    const errorMsg = data && data.message ? data.message : `Ошибка HTTP ${response.status}`;
    throw new Error(errorMsg);
  }

  return data as T;
}

function hashPassword(password) {
  return crypto.createHash('sha256').update(password).digest('hex');

// Добавляет колонку в таблицу при необходимости (мягкая миграция)
async function addColumnIfMissing(table, column, definition) {
  const columns = await db.all(`PRAGMA table_info(${table})`);
  const exists = columns.some((c) => c.name === column);
  if (!exists) {
    await db.exec(`ALTER TABLE ${table} ADD COLUMN ${column} ${definition}`);
  }
}

const initDb = async () => {
  try {

      db = await open({
        filename: path.join(__dirname, 'db.sqlite'),
        driver: sqlite3.Database
      });

      await db.exec('PRAGMA foreign_keys = ON;');
    }

    // 1. Подразделения (departments)
    await db.exec(`
      CREATE TABLE IF NOT EXISTS departments (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL UNIQUE,
        code TEXT NOT NULL UNIQUE,
        headName TEXT
      )
    `);

    await db.exec(`
      CREATE TABLE IF NOT EXISTS locations (

      if (!log) return 0;

      const dateToSet = endDate || (status === 'Завершено' ? new Date().toISOString().slice(0, 10) : null);
      const res = await getDb().run(
        'UPDATE maintenance_logs SET status = ?, endDate = COALESCE(?, endDate) WHERE id = ?',

        dateToSet,
        id
      );

const { verifyAccessToken, verifyRefreshToken, generateAccessToken } = require('../utils/jwt');
const { getUserIdByToken } = require('../db/tokens');
const { getUserById } = require('../db/users');
const {
  COOKIE_ACCESS_NAME,
  COOKIE_REFRESH_NAME,
  ACCESS_COOKIE_MAX_AGE
} = require('../constants');

async function authMiddleware(req, res, next) {
  try {
    const accessToken = req.cookies[COOKIE_ACCESS_NAME];

    // 1. Быстрый путь: Access Token валиден (без обращения к БД)
    if (accessToken) {
      const { valid, decoded } = verifyAccessToken(accessToken);
      if (valid && decoded) {
        req.userId = decoded.id;
        req.userLogin = decoded.login;
        req.userRole = decoded.role;
        return next();
      }
    }

    // 2. Если Access Token истек или отсутствует — проверяем Refresh Token
    const refreshToken = req.cookies[COOKIE_REFRESH_NAME];
    if (!refreshToken) {
      return res.status(401).json({ message: 'Пользователь не авторизован' });
    }

    const { valid: rfValid, decoded: rfDecoded } = verifyRefreshToken(refreshToken);
    if (!rfValid || !rfDecoded) {
      return res.status(401).json({ message: 'Срок сессии истек. Пожалуйста, выполните вход' });
    }

    // Проверяем наличие refresh-токена в базе (на случай отзыва сессии)
    const tokenUserId = await getUserIdByToken(refreshToken);
    if (!tokenUserId || tokenUserId !== rfDecoded.id) {
      return res.status(401).json({ message: 'Сессия недействительна' });
    }

    const user = await getUserById(tokenUserId);
    if (!user) {
      return res.status(401).json({ message: 'Пользователь не найден' });
    }

    // Автоматически перевыпускаем Access Token в куку
    const newAccessToken = generateAccessToken(user);
    res.cookie(COOKIE_ACCESS_NAME, newAccessToken, {
      maxAge: ACCESS_COOKIE_MAX_AGE,
      httpOnly: true,
      sameSite: 'lax'
    });

    req.userId = user.id;
    req.userLogin = user.login;
    req.userRole = user.role;
    next();
  } catch (error) {
    console.error('authMiddleware error:', error);
    res.status(500).json({ message: 'Ошибка при проверке авторизации' });
  }
}

module.exports = authMiddleware;

// Пропускает запрос дальше, только если роль пользователя входит в список разрешённых.
// Использовать ПОСЛЕ authMiddleware (он кладёт роль в req.userRole).
//
// Пример: router.post('/', requireRole('manager', 'admin'), handler)
//
// 401 — «мы не знаем, кто вы» (это делает authMiddleware)
// 403 — «мы знаем, кто вы, но вам нельзя» (это делает этот middleware)
function requireRole(...allowedRoles) {
  return (req, res, next) => {
    if (!allowedRoles.includes(req.userRole)) {
      return res.status(403).json({ message: 'Недостаточно прав для этого действия' });
    }
    next();
  };
}

authRouter.post('/login', async (req, res) => {
  try {
    const { login, password } = req.body || {};

    if (!login || !password) {
      return res.status(400).json({ message: 'Введите логин и пароль' });
    }

    const user = await getUserByLogin(String(login).trim());

    if (!user || user.password !== hashPassword(String(password))) {

    }

    const accessToken = generateAccessToken(user);
    const refreshToken = generateRefreshToken(user);

    // Сохраняем refresh-токен в базу данных
    await saveRefreshToken(user.id, refreshToken);

    // Устанавливаем HTTP-only куки

    });

    res.cookie(COOKIE_REFRESH_NAME, refreshToken, {
      maxAge: REFRESH_COOKIE_MAX_AGE,
      httpOnly: true,
      sameSite: 'lax'
    });

    res.status(200).json({
      ok: true,
      user: { id: user.id, login: user.login, role: user.role }
    });
  } catch (error) {
    console.error('Login error:', error);

authRouter.post('/refresh', async (req, res) => {

    const refreshToken = req.cookies[COOKIE_REFRESH_NAME];

    if (!refreshToken) {
      return res.status(401).json({ message: 'Refresh-токен отсутствует' });
    }

    const { valid, decoded } = verifyRefreshToken(refreshToken);
    if (!valid || !decoded) {
      return res.status(401).json({ message: 'Срок действия сессии истек' });
    }

    const userId = await getUserIdByToken(refreshToken);
    if (!userId || userId !== decoded.id) {

    }

    const user = await getUserById(userId);
    if (!user) {
      return res.status(401).json({ message: 'Пользователь не найден' });
    }

    const newAccessToken = generateAccessToken(user);
    const newRefreshToken = generateRefreshToken(user);

    await saveRefreshToken(user.id, newRefreshToken);

    res.cookie(COOKIE_ACCESS_NAME, newAccessToken, {
      maxAge: ACCESS_COOKIE_MAX_AGE,
      httpOnly: true,
      sameSite: 'lax'
    });

    res.cookie(COOKIE_REFRESH_NAME, newRefreshToken, {

    });

      user: { id: user.id, login: user.login, role: user.role }
    });
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при обновлении токена' });
  }

const express = require('express');
const { getDb } = require('../db/db');
const authMiddleware = require('../middleware/auth');
const requireRole = require('../middleware/requireRole');
const { REQUEST_TYPES, REQUEST_PRIORITIES, REQUEST_STATUSES } = require('../constants');

const requestsRouter = express.Router();
requestsRouter.use(authMiddleware);

// Получить список заявок (с присоединением инв. номера и наименования оборудования)
requestsRouter.get('/', async (req, res) => {
  try {
    let query = `
      SELECT r.*, e.name AS equipmentName, e.invNumber
      FROM requests r
      LEFT JOIN equipment e ON r.equipmentId = e.id
    `;
    const params = [];

    if (req.userRole === 'employee') {
      query += ` WHERE r.applicant = ?`;
      params.push(req.userLogin);
    }

    query += ` ORDER BY r.id DESC`;
    const items = await getDb().all(query, ...params);
    res.json(items);
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при получении списка заявок' });
  }
});

// Создать заявку с возможностью указания оборудования
requestsRouter.post('/', async (req, res) => {
  try {
    const { type, title, applicant, priority, equipmentId } = req.body || {};

    if (!REQUEST_TYPES.includes(type)) {
      return res.status(400).json({ message: `Недопустимый тип заявки: ${REQUEST_TYPES.join(', ')}` });
    }
    if (!title || !String(title).trim()) {
      return res.status(400).json({ message: 'Укажите тему заявки' });
    }

    const cleanPriority = priority || 'Средний';
    if (!REQUEST_PRIORITIES.includes(cleanPriority)) {
      return res.status(400).json({ message: 'Недопустимый приоритет заявки' });
    }

    let cleanApplicant = req.userLogin;
    if (req.userRole !== 'employee' && applicant && String(applicant).trim()) {
      cleanApplicant = String(applicant).trim();
    }

    const result = await getDb().run(`
      INSERT INTO requests (type, title, applicant, equipmentId, priority, status, createdAt)
      VALUES (?, ?, ?, ?, ?, 'На согласовании', datetime('now', 'localtime'))
    `,
      type,
      String(title).trim(),
      cleanApplicant,
      equipmentId ? Number(equipmentId) : null,
      cleanPriority
    );

    const created = await getDb().get('SELECT * FROM requests WHERE id = ?', result.lastID);
    res.status(201).json(created);
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при создании заявки' });
  }
});

// Смена статуса заявки
requestsRouter.patch('/:id', requireRole('manager', 'admin'), async (req, res) => {
  try {
    const id = Number(req.params.id);
    const { status } = req.body || {};
    if (!REQUEST_STATUSES.includes(status)) {
      return res.status(400).json({ message: 'Недопустимый статус заявки' });
    }

    const result = await getDb().run('UPDATE requests SET status = ? WHERE id = ?', status, id);
    if (!result.changes) return res.status(404).json({ message: 'Заявка не найдена' });

    res.json({ id, status });
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при обновлении заявки' });
  }
});

module.exports = requestsRouter;

const express = require('express');

// Профиль текущего пользователя (нужна авторизация)

    res.status(200).json(user);
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при получении профиля' });
  }

// Регистрация нового пользователя (открытый эндпоинт).
// Роль выдаётся всегда «employee»: выбрать себе роль при регистрации нельзя.
userRouter.post('/register', async (req, res) => {
  try {
    const { login, password } = req.body || {};

    if (!login || !password) {
      return res.status(400).json({ message: 'Введите логин и пароль' });
    }

    const cleanLogin = String(login).trim();

    if (cleanLogin.length < 3) {
      return res.status(400).json({ message: 'Логин должен содержать не менее 3 символов' });
    }

    if (String(password).length < 6) {
      return res.status(400).json({ message: 'Пароль должен содержать не менее 6 символов' });
    }

    const user = await getUserByLogin(cleanLogin);
    if (user) {
      return res.status(400).json({ message: 'Такой пользователь уже есть' });
    }

    const newUser = await addUser(cleanLogin, String(password), 'employee');
    res.status(201).json(newUser);
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при регистрации пользователя' });
  }
});

// Список всех пользователей (только администратор)
userRouter.get('/', authMiddleware, requireRole('admin'), async (req, res) => {
  try {
    const users = await getUsers();
    res.json(users);
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при получении списка пользователей' });
  }
});

// Смена роли пользователя (только администратор)
userRouter.patch('/:id/role', authMiddleware, requireRole('admin'), async (req, res) => {
  try {
    const id = Number(req.params.id);
    if (!Number.isInteger(id) || id <= 0) {
      return res.status(400).json({ message: 'Некорректный id' });
    }

    const { role } = req.body || {};
    if (!USER_ROLES.includes(role)) {
      return res
        .status(400)
        .json({ message: `Недопустимая роль. Допустимо: ${USER_ROLES.join(', ')}` });
    }

    // Чтобы администратор случайно не лишил себя прав и не оставил систему без админа
    if (id === req.userId) {
      return res.status(400).json({ message: 'Нельзя изменить собственную роль' });
    }

    const changes = await updateUserRole(id, role);
    if (changes === 0) {
      return res.status(404).json({ message: 'Пользователь не найден' });
    }

    res.json({ id, role });
  } catch (error) {
    res.status(500).json({ message: 'Ошибка при смене роли' });
  }
});

userRouter.delete('/:id', authMiddleware, requireRole('admin'), async (req, res) => {

  } catch (error) {
    res.status(500).json({ message: 'Ошибка при удалении пользователя' });
  }
});

const jwt = require('jsonwebtoken');
const {
  JWT_ACCESS_SECRET,
  JWT_REFRESH_SECRET,
  ACCESS_TOKEN_EXPIRES_IN,
  REFRESH_TOKEN_EXPIRES_IN
} = require('../constants');

// Генерация короткоживущего токена доступа (15 минут)
function generateAccessToken(user) {
  return jwt.sign(
    {
      id: user.id,
      login: user.login,
      role: user.role
    },
    JWT_ACCESS_SECRET,
    { expiresIn: ACCESS_TOKEN_EXPIRES_IN }
  );
}

// Генерация долгоживущего refresh-токена (7 дней)
function generateRefreshToken(user) {
  return jwt.sign(
    {
      id: user.id,
      login: user.login
    },
    JWT_REFRESH_SECRET,
    { expiresIn: REFRESH_TOKEN_EXPIRES_IN }
  );
}

// Проверка access-токена
function verifyAccessToken(token) {
  try {
    return { valid: true, decoded: jwt.verify(token, JWT_ACCESS_SECRET) };
  } catch (error) {
    return { valid: false, expired: error.name === 'TokenExpiredError', error };
  }
}

// Проверка refresh-токена
function verifyRefreshToken(token) {
  try {
    return { valid: true, decoded: jwt.verify(token, JWT_REFRESH_SECRET) };
  } catch (error) {
    return { valid: false, expired: error.name === 'TokenExpiredError', error };
  }
}

module.exports = {
  generateAccessToken,
  generateRefreshToken,
  verifyAccessToken,
  verifyRefreshToken
};
